cmake_minimum_required(VERSION 3.16)
project(rover_state)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json REQUIRED)
set(dependencies arm_task_interfaces rcl_interfaces rclcpp rclcpp_action nav2_msgs nav_msgs geometry_msgs std_msgs std_srvs tf2_ros tf2_geometry_msgs)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(mission_core src/mission_node.cpp src/centered_grasp.cpp src/cave_turnaround.cpp)
target_include_directories(mission_core PUBLIC include)
target_link_libraries(mission_core nlohmann_json::nlohmann_json)
ament_target_dependencies(mission_core ${dependencies})
add_executable(mission_state src/main.cpp)
target_link_libraries(mission_state mission_core)
add_executable(mission_motion_gate src/motion_gate.cpp)
target_include_directories(mission_motion_gate PRIVATE include)
target_link_libraries(mission_motion_gate nlohmann_json::nlohmann_json)
ament_target_dependencies(mission_motion_gate rclcpp geometry_msgs std_msgs)
foreach(target mission_core mission_state mission_motion_gate)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS mission_state mission_motion_gate DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/standalone_grasp_executor.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
